<firestore_rules_block>
Per D-28, exact contents of firestore.rules:

```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{uid}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    match /system/{doc} {
      allow read: if request.auth != null;
      allow write: if false;
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
```

The trailing match-all deny is defense-in-depth; Firestore default-deny covers it but explicit is better.
</firestore_rules_block>

```ts
import { auth } from './firebase';

export async function api(path: string, init: RequestInit = {}): Promise<Response> {
  if (!path.startsWith('/api/')) throw new Error('api() only handles /api/* paths');
  const user = auth.currentUser;
  const token = user ? await user.getIdToken() : null;
  const headers = new Headers(init.headers);
  if (token) headers.set('Authorization', `Bearer ${token}`);
  if (init.body && !headers.has('Content-Type')) headers.set('Content-Type', 'application/json');
  return fetch(path, { ...init, headers });
}
```

import { useEffect, useState } from 'react';
import { doc, onSnapshot, type DocumentData } from 'firebase/firestore';
import { db } from '../lib/firebase';

export interface UserDocShape {
  email?: string;
  displayName?: string;
  activeBrandId?: string;
  apiKeys?: { anthropic?: string; metaGraph?: string };
}

export function useUserDoc(uid: string | null): { data: UserDocShape | null; loading: boolean } {
  const [data, setData] = useState&lt;UserDocShape | null&gt;(null);
  const [loading, setLoading] = useState(true);
  useEffect(() =&gt; {
    if (!uid) { setData(null); setLoading(false); return; }

    const unsub = onSnapshot(doc(db, 'users', uid), (snap) =&gt; {
      setData((snap.data() as UserDocShape) ?? null);
      setLoading(false);
    });
    return () =&gt; unsub();
  }, [uid]);
  return { data, loading };
