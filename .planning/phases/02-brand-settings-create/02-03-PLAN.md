streamGenerate signature:
```ts
export interface StreamGenerateOptions {
  token: string;                               // Firebase ID token
  body: GenerateRequest;                       // from 02-02 shared/schemas/generateRequest.ts
  signal: AbortSignal;
  onChunk: (text: string) => void;             // raw text from {type:'chunk'} events, concatenated by caller
  onComplete: (payload: { postId: string; slides: SocialSlide[]; caption: string }) => void;
  onError: (err: { message: string; code?: string }) => void;
}
export async function streamGenerate(opts: StreamGenerateOptions): Promise<void>;
```
Implementation skeleton (the executor must match this loop precisely - per 02-RESEARCH.md Pitfall #5):
```ts
const res = await fetch('/api/generate', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${opts.token}`,
  },
  body: JSON.stringify(opts.body),
  signal: opts.signal,
});
if (!res.ok) {
  const errBody = await res.json().catch(() => ({ error: res.statusText }));
  opts.onError({ message: errBody.error ?? 'Request failed', code: errBody.code });
  return;
}
const reader = res.body!.getReader();
const decoder = new TextDecoder('utf-8');
let buf = '';
while (true) {
  const { value, done } = await reader.read();
  if (value) buf += decoder.decode(value, { stream: true });
  let nl: number;
  while ((nl = buf.indexOf('\n')) !== -1) {
    const line = buf.slice(0, nl).trim();
    buf = buf.slice(nl + 1);
    if (!line) continue;
    const evt = JSON.parse(line);
    if (evt.type === 'chunk') opts.onChunk(evt.text);
    else if (evt.type === 'complete') opts.onComplete(evt);
    else if (evt.type === 'error') opts.onError({ message: evt.error, code: evt.code });
  }
  if (done) {
    if (buf.trim()) {
      // last line without trailing \n - still parse
      const evt = JSON.parse(buf.trim());
      if (evt.type === 'complete') opts.onComplete(evt);
      else if (evt.type === 'error') opts.onError({ message: evt.error, code: evt.code });
    }
    return;
  }
}
```
